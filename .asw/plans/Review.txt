// Frecency tracking (not just MRU)
AppUsageTracker {
    property var appLaunches: ({})  // { appId: { count, lastLaunchTime, score } }

    function recordLaunch(appId: string) {
        const now = Date.now();
        if (!appLaunches[appId]) {
            appLaunches[appId] = { count: 0, lastLaunchTime: 0, score: 0 };
        }
        appLaunches[appId].count++;
        appLaunches[appId].lastLaunchTime = now;
        // Frecency score = frequency * time_decay
        appLaunches[appId].score = appLaunches[appId].count * Math.exp(-0.001 * (now - appLaunches[appId].lastLaunchTime));
    }
}

    property bool visible: window.hovered || window.focused

    Repeater {
        model: ["left", "right", "top", "bottom", "top-left", "top-right", "bottom-left", "bottom-right"]

        MouseArea {
            anchors.fill: parent
            cursorShape: {
                if (modelData === "left" || modelData === "right") return Qt.SizeHorCursor;
                if (modelData === "top" || modelData === "bottom") return Qt.SizeVerCursor;
                // Diagonal
                return Qt.SizeFDiagCursor;
            }

            onMouseXChanged: {
                if (pressed) {
                    const delta = mouseX - lastX;
                    lastX = mouseX;

                    if (modelData.includes("left")) {
                        window.x += delta;
                        window.width -= delta;
                    } else if (modelData.includes("right")) {
                        window.width += delta;
                    }
                }
            }
        }
    }

// core/DynamicTips.qml additions
TipHistory {
    property var dismissedTips: QSettings.value("tips/dismissed", [])
    property var viewedTips: QSettings.value("tips/viewed", {})  // { tipId: count }

    function shouldShowTip(tip: Object): bool {
        if (dismissedTips.includes(tip.id)) return false;
        if ((viewedTips[tip.id] ?? 0) > 2) return false;  // Max 2 showings
        if (tip.requiresCondition && !tip.condition()) return false;
        return true;
    }

    function recordViewed(tipId: string) {
        viewedTips[tipId] = (viewedTips[tipId] ?? 0) + 1;
        QSettings.setValue("tips/viewed", viewedTips);
    }

    function recordDismissed(tipId: string) {
        dismissedTips.push(tipId);
        QSettings.setValue("tips/dismissed", dismissedTips);
    }
}
